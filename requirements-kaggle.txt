# Kaggle-only requirements for the judge committee runner (purva/committee/).
#
# Kaggle's T4x2 image ships CUDA 12 drivers. vllm's latest releases (>=0.7)
# pull a CUDA-13 build (torch wheels depending on libnvrtc.so.13), which
# crashes on Kaggle T4 workers with ImportError: libnvrtc.so.13. Pin vllm to
# a CUDA-12.1-compatible release, and pin torch/transformers to the versions
# that release expects — this intentionally overrides Kaggle's preinstalled
# torch build rather than relying on it, since that preinstalled build is
# not guaranteed to match vllm's pinned CUDA-12 wheel.

vllm==0.26.0
torch==2.13.0
transformers==5.5.0
accelerate>=0.33
huggingface_hub>=0.24

# vllm 0.6.6.post1's xgrammar integration calls the low-level C++ binding
# xgr_core.TokenizerInfo.from_huggingface directly. xgrammar removed that
# method from the core binding in 0.1.15 (moved to the pure-Python wrapper),
# which crashes vllm 0.6.6.post1 with AttributeError: type object
# 'TokenizerInfo' has no attribute 'from_huggingface' the moment guided
# decoding is used (confirmed against a real Kaggle run; see
# vllm-project/vllm#16500). vllm's own constraint is only xgrammar>=0.1.6,
# so an unpinned install grabs whatever is latest and breaks. Pin below 0.1.15.
xgrammar==0.1.32

# quantization backends used by purva/committee/models.py REGISTRY entries
autoawq>=0.2.6      # for the Qwen2.5-14B-Instruct-AWQ judge
bitsandbytes>=0.43  # for the remaining judges (4-bit)
